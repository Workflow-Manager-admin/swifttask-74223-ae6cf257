# 1. Open your terminal and go to the frontend folder
cd todo_frontend_app_workspace/todo_frontend_app

# 2. Install dependencies
npm install

# 3. Configure the frontend to talk to the backend
echo "VITE_TODO_API_BASE=http://localhost:5000" > .env.local

# 4. Start the Vue development server
npm run dev

# The app will be available at http://localhost:3000
# All requests will be sent to the Flask backend at http://localhost:5000

# To stop the server, use Ctrl+C.
# If you get “npm: command not found”, please install Node.js from https://nodejs.org/
